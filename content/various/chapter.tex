\begin{itemize}
		\item \verb@signal(SIGSEGV, [](int) { _Exit(0); });@ biến segfault thành Wrong Answer.
			Tương tự, có thể bắt SIGABRT (assertion thất bại) và SIGFPE (chia cho không).
			Lỗi của \verb@_GLIBCXX_DEBUG@ sinh ra SIGABRT (hoặc SIGSEGV trên gcc 5.4.0).
		\item \verb@feenableexcept(29);@ dừng chương trình khi gặp NaN (\texttt 1), chia cho 0 (\texttt 4), vô cực (\texttt 8) và số không chuẩn hóa (denormal) (\texttt{16}).
	\end{itemize}

\section{Mẹo tối ưu}
	\verb@__builtin_ia32_ldmxcsr(40896);@ tắt denormal (denormal làm số thực chậm hơn 20 lần khi gần giá trị nhỏ nhất).
	\subsection{Thủ thuật bit}
		\begin{itemize}
			\item \verb@x & -x@ là bit thấp nhất đang bật của \texttt{x}.
			\item \verb@for (int x = m; x; ) { --x &= m; ... }@ duyệt qua mọi mặt nạ con của \texttt{m} (trừ chính \texttt{m}).
			\item \verb@c = x&-x, r = x+c; (((r^x) >> 2)/c) | r@ là số kế tiếp sau \texttt{x} có cùng số bit được bật.
			\item \verb@rep(b,0,K) rep(i,0,(1 << K))@ \\ \verb@  if (i & 1 << b) D[i] += D[i^(1 << b)];@ tính tổng trên mọi tập con.
		\end{itemize}
	\subsection{Pragma}
		\begin{itemize}
			\item \lstinline{#pragma GCC optimize ("Ofast")} giúp GCC tự động vector hóa vòng lặp và tối ưu số thực tốt hơn.
			\item \lstinline{#pragma GCC target ("avx2")} có thể tăng gấp đôi hiệu năng của code đã vector hóa, nhưng gây crash trên máy cũ.
			\item \lstinline{#pragma GCC optimize ("trapv")} dừng chương trình khi tràn số nguyên (nhưng rất chậm).
		\end{itemize}
