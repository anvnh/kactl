```sh
  texlive-geometry \
  texlive-tools \
  texlive-fancyhdr \

  texlive-amsfonts \
  texlive-tocloft \
  texlive-titlesec \
  texlive-psnfss \
  texlive-graphics \
  texlive-paralist \

  texlive-ulem \
  texlive-hyperref \
  texlive-microtype \
  texlive-collection-fontsrecommended \
  texlive-dvips \

  texlive-cm
```
